"use client";

import { useCallback, useEffect, useState } from "react";
import { configurationService } from "../services/configuration.service";
import { SETTING_DEFINITIONS } from "../types/configuration.types";
import type { BranchConfiguration } from "../types/configuration.types";

export function useBranchConfiguration(branchId: number | null) {
  const [settings, setSettings] = useState<BranchConfiguration[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [savingKey, setSavingKey] = useState<string | null>(null);

  const load = useCallback(async () => {
    if (branchId == null) {
      setSettings([]);
      setLoading(false);
      return;
    }
    setLoading(true);
    setError(null);
    try {
      const data = await configurationService.getByBranch(branchId);
      setSettings(data);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "No se pudo cargar la configuraciÃ³n de la sede",
      );
    } finally {
      setLoading(false);
    }
  }, [branchId]);

  useEffect(() => {
    load();
  }, [load]);

  const getValue = useCallback(
    (key: string) => settings.find((s) => s.key === key)?.value ?? "",
    [settings],
  );

  // Decide POST vs PUT automÃ¡ticamente: si la clave ya existe para esta
  // sede, actualiza; si no, la crea. AsÃ­ el formulario no tiene que saber
  // cuÃ¡les claves ya tienen fila en la BD y cuÃ¡les no.
  const saveSetting = useCallback(
    async (key: string, value: string, dataType: string) => {
      if (branchId == null) return;
      setSavingKey(key);
      try {
        const existing = settings.find((s) => s.key === key);
        if (existing) {
          const updated = await configurationService.update(existing.id, {
            value,
            dataType,
            description: existing.description ?? undefined,
          });
          setSettings((prev) =>
            prev.map((s) => (s.id === updated.id ? updated : s)),
          );
        } else {
          const definition = SETTING_DEFINITIONS.find((d) => d.key === key);
          const created = await configurationService.create({
            branchId,
            key,
            value,
            dataType,
            description: definition?.label,
          });
          setSettings((prev) => [...prev, created]);
        }
      } finally {
        setSavingKey(null);
      }
    },
    [branchId, settings],
  );

  return {
    settings,
    loading,
    error,
    savingKey,
    getValue,
    saveSetting,
    reload: load,
  };
}


===== .\features\Configurations\services\configuration.service.ts =====
import { http } from "@/config/api";
import type {
  BranchConfiguration,
  CreateConfigurationPayload,
  UpdateConfigurationPayload,
} from "../types/configuration.types";

export const configurationService = {
  // GET /api/Configurations/branch/{branchId}
  getByBranch: async (branchId: number): Promise<BranchConfiguration[]> => {
    const { data } = await http.get<BranchConfiguration[]>(
      `/Configurations/branch/${branchId}`,
    );
    return data;
  },

  // POST /api/Configurations
  create: async (
    payload: CreateConfigurationPayload,
  ): Promise<BranchConfiguration> => {
    const { data } = await http.post<BranchConfiguration>(
      "/Configurations",
      payload,
    );
    return data;
  },

  // PUT /api/Configurations/{id}
  update: async (
    id: number,
    payload: UpdateConfigurationPayload,
  ): Promise<BranchConfiguration> => {
    const { data } = await http.put<BranchConfiguration>(
      `/Configurations/${id}`,
      payload,
    );
    return data;
  },
};

let redirecting = false;

// --- LÃ³gica de refresh automÃ¡tico ---
let isRefreshing = false;
let refreshSubscribers: Array<{
  resolve: (token: string) => void;
  reject: (err: unknown) => void;
}> = [];

function onTokenRefreshed(newToken: string) {
  refreshSubscribers.forEach((s) => s.resolve(newToken));
  refreshSubscribers = [];
}

function onRefreshFailed(err: unknown) {
  refreshSubscribers.forEach((s) => s.reject(err));
  refreshSubscribers = [];
}

function forceLogout() {
  if (redirecting) return;
  redirecting = true;
  localStorage.removeItem(SESSION_KEY);
  showSessionExpiredToast();
  setTimeout(() => {
    window.location.href = "/login";
  }, 2500);
}

http.interceptors.response.use(
  (response) => response,
  async (error) => {
    // Reescribe error.message con el mensaje real que manda el backend
    // (ej. "Ya existe una mesa con ese nÃºmero en esta sucursal"), para que
    // cualquier `catch` en cualquier hook lo reciba correcto automÃ¡ticamente,
    // en vez del genÃ©rico "Request failed with status code 400" de Axios.
    const backendMessage =
      error?.response?.data?.error ?? error?.response?.data?.message;
    if (backendMessage) {
      error.message = backendMessage;
    }

    if (typeof window === "undefined") {
      return Promise.reject(error);
    }

    const originalRequest = error.config;
    const isLoginAttempt = originalRequest?.url?.includes("/Auth/login");
    const isRefreshAttempt = originalRequest?.url?.includes(
      "/Auth/refresh-token",
    );

    if (error?.response?.status !== 401 || isLoginAttempt || isRefreshAttempt) {
      // Si el propio refresh-token fallÃ³ (refresh token vencido/invÃ¡lido),
      // no hay nada mÃ¡s que hacer: cerramos sesiÃ³n.
      if (error?.response?.status === 401 && isRefreshAttempt) {
        forceLogout();
      }
      return Promise.reject(error);
    }

    // Evita reintentar infinitamente la misma peticiÃ³n
    if (originalRequest._retry) {
      forceLogout();
      return Promise.reject(error);
    }
    originalRequest._retry = true;

    const session = readSession();
    if (!session?.refreshToken) {
      forceLogout();
      return Promise.reject(error);
    }

    // Si ya hay un refresh en curso, encolamos esta peticiÃ³n
    // hasta que termine, en vez de disparar refresh-token varias veces a la vez.
    if (isRefreshing) {
      return new Promise((resolve, reject) => {
        refreshSubscribers.push({
          resolve: (newToken: string) => {
            originalRequest.headers.Authorization = `Bearer ${newToken}`;
            resolve(http(originalRequest));
          },
          reject,
        });
      });
    }

    isRefreshing = true;

    try {
      const { data } = await axios.post(
        `${http.defaults.baseURL}/Auth/refresh-token`,
        { refreshToken: session.refreshToken },
      );

      const newSession = {
        ...session,
        userId: data.userId ?? session.userId,
        identifier: data.identifier,
        roles: data.roles,
        token: data.token,
        refreshToken: data.refreshToken,
      };
      saveSession(newSession);

      isRefreshing = false;
      onTokenRefreshed(data.token);

      originalRequest.headers.Authorization = `Bearer ${data.token}`;
      return http(originalRequest);
    } catch (refreshError) {
      isRefreshing = false;
      onRefreshFailed(refreshError);
      forceLogout();
      return Promise.reject(refreshError);
    }
  },
);
